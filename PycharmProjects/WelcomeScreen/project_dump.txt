_ORIGINAL_GET_FRAME = FFMPEG_AudioReader.get_frame


def _safe_get_frame(self: FFMPEG_AudioReader, tt: Any) -> np.ndarray:
    try:
        return _ORIGINAL_GET_FRAME(self, tt)
    except (OSError, IOError):
        if isinstance(tt, np.ndarray):
            return np.zeros((len(tt), self.nchannels), dtype=np.float64)
        return np.zeros((self.nchannels,), dtype=np.float64)

def get_font(size: int) -> ImageFont.FreeTypeFont:
    for path in (
        "/System/Library/Fonts/Supplemental/Arial Bold.ttf",
        "/System/Library/Fonts/Supplemental/Impact.ttf",
        "/System/Library/Fonts/Helvetica.ttc",
    ):
        if os.path.exists(path):
            return ImageFont.truetype(path, size)
    return ImageFont.load_default()


def wrapped(text: str, width: int = 23) -> str:
    return "\n".join(textwrap.wrap(text.upper(), width=width, break_long_words=False))


def get_scene_duration(scene: dict[str, Any]) -> float:
    timing = str(scene.get("timing", "0-5")).replace(" ", "").split("-", 1)
    try:
        return max(float(timing[1]) - float(timing[0]), 0.1)
    except (IndexError, ValueError):
        return 5.0

def _rgb(val: Any, default: tuple[int, int, int] = (200, 200, 200)) -> tuple[int, int, int]:
    if isinstance(val, (list, tuple)) and len(val) >= 3:
        return tuple(max(0, min(255, int(val[index]))) for index in range(3))
    return default


def _draw_actor(draw: ImageDraw.ImageDraw, cmd: dict[str, Any], dx: int, dy: int) -> None:
    x = int(cmd.get("x", 540)) + dx
    y = int(cmd.get("y", 950)) + dy
    scale = float(cmd.get("scale", 1.0))
    shirt = _rgb(cmd.get("shirt"), (62, 118, 176))
    skin = _rgb(cmd.get("skin"), (232, 202, 175))
    hair = _rgb(cmd.get("hair"), (48, 36, 30))
    outline = (20, 24, 30)
    emotion = str(cmd.get("emotion", "scared"))
    pose = str(cmd.get("pose", "stand"))

    if pose == "lie":
        draw.rounded_rectangle((x - int(180 * scale), y, x + int(140 * scale), y + int(95 * scale)),
                               radius=int(24 * scale), fill=shirt, outline=outline, width=5)
        draw.ellipse((x - int(250 * scale), y - int(10 * scale),
                      x - int(150 * scale), y + int(90 * scale)),
                     fill=skin, outline=outline, width=5)
        return

    body_w, body_h = int(95 * scale), int(230 * scale)
    draw.rounded_rectangle((x - body_w, y, x + body_w, y + body_h),
                           radius=int(32 * scale), fill=shirt, outline=outline, width=5)
    leg_w, leg_h = int(36 * scale), int(175 * scale)
    for left, right in (
        (x - body_w + int(14 * scale), x - int(8 * scale)),
        (x + int(8 * scale), x + body_w - int(14 * scale)),
    ):
        draw.rounded_rectangle((left, y + body_h - int(15 * scale), right, y + body_h + leg_h),
                               radius=int(14 * scale), fill=(38, 44, 54), outline=outline, width=5)

    radius = int(68 * scale)
    head_y = y - radius - int(8 * scale)
    if cmd.get("hood"):
        draw.ellipse((x - radius - int(16 * scale), head_y - int(16 * scale),
                      x + radius + int(16 * scale), head_y + radius + int(16 * scale)),
                     fill=shirt, outline=outline, width=5)
    draw.ellipse((x - radius, head_y - radius, x + radius, head_y + radius),
                 fill=skin, outline=outline, width=5)
    draw.pieslice((x - radius, head_y - radius, x + radius, head_y + radius),
                  180, 360, fill=hair)

    eye_radius = int(13 * scale) if emotion in ("scared", "shock") else int(9 * scale)
    for eye_x in (x - int(24 * scale), x + int(24 * scale)):
        draw.ellipse((eye_x - eye_radius, head_y - eye_radius,
                      eye_x + eye_radius, head_y + eye_radius),
                     fill="white", outline=outline, width=3)
        draw.ellipse((eye_x - int(5 * scale), head_y - int(5 * scale),
                      eye_x + int(5 * scale), head_y + int(5 * scale)), fill=outline)
    if emotion in ("scared", "shock"):
        draw.ellipse((x - int(12 * scale), head_y + int(22 * scale),
                      x + int(12 * scale), head_y + int(44 * scale)),
                     fill=(30, 20, 25), outline=outline, width=3)
    else:
        draw.line((x - int(16 * scale), head_y + int(30 * scale),
                   x + int(16 * scale), head_y + int(30 * scale)), fill=outline, width=4)

    if pose == "cover_head":
        for left, right in (
            (x - body_w - int(25 * scale), x - body_w + int(15 * scale)),
            (x + body_w - int(15 * scale), x + body_w + int(25 * scale)),
        ):
            draw.rounded_rectangle((left, head_y - int(20 * scale), right, y + int(90 * scale)),
                                   radius=int(16 * scale), fill=shirt, outline=outline, width=4)
    elif pose == "phone":
        draw.rounded_rectangle((x + body_w - int(20 * scale), head_y,
                                x + body_w + int(35 * scale), y + int(110 * scale)),
                               radius=int(16 * scale), fill=shirt, outline=outline, width=4)
        draw.rounded_rectangle((x + radius - int(5 * scale), head_y - int(25 * scale),
                                x + radius + int(28 * scale), head_y + int(35 * scale)),
                               radius=6, fill=(18, 20, 24), outline=(90, 95, 105), width=2)


def draw_character(draw: ImageDraw.ImageDraw, x: int, y: int, scale: float = 1.0,
                   coat: tuple[int, int, int] = (62, 118, 176),
                   skin: tuple[int, int, int] = (232, 202, 175),
                   hair: tuple[int, int, int] = (48, 36, 30),
                   emotion: str = "scared", pose: str = "stand",
                   hood: bool = False, name_tag: str = "") -> None:
    _draw_actor(draw, {
        "x": x, "y": y, "scale": scale, "shirt": coat, "skin": skin,
        "hair": hair, "emotion": emotion, "pose": pose, "hood": hood,
        "name_tag": name_tag,
    }, 0, 0)


def draw_city_skyline(draw: ImageDraw.ImageDraw, y_base: int = 1150,

    buildings = (
        (30, 680, 140), (155, 590, 165), (305, 730, 130),

    )
    for index, (x, y, width) in enumerate(buildings):
        color = (68, 76, 92) if flash else ((26, 32, 44) if index % 2 == 0 else (18, 23, 32))
        draw.rectangle((x, y, x + width, y_base), fill=color, outline=(12, 16, 22), width=3)
        for window_y in range(y + 35, y_base - 60, 65):
            for window_x in range(x + 22, x + width - 35, 48):
                if (window_x + window_y + index) % 5 == 0:
                    draw.rectangle((window_x, window_y, window_x + 22, window_y + 32),
                                   fill=(235, 205, 115))


def draw_environment_details(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    story_meta: dict[str, Any],
    phase: float,
    accent: tuple[int, int, int],
) -> None:
    """Adds generated location props to the procedural background."""
    description = " ".join(
        [str(story_meta.get("location", ""))] +
        [
            str(scene.get(key, ""))
            for key in ("location", "environment_detail",
                        "background_scene", "background_objects")
        ]
    ).lower()
    shadow, metal, warm = (18, 22, 28), (125, 136, 148), (235, 185, 72)

    if any(word in description for word in ("маяк", "lighthouse", "берег", "shore")):
        draw.rectangle((790, 470, 900, 1040), fill=shadow, outline=metal, width=8)
        draw.polygon([(745, 470), (945, 470), (900, 380), (790, 380)],
                     fill=shadow, outline=metal)
        draw.ellipse((775, 420, 915, 560), outline=warm, width=8)
        draw.line((845, 490, 845 + int(90 * math.sin(phase)), 490),
                  fill=warm, width=5)
    elif any(word in description for word in ("метро", "туннел", "subway", "station", "перрон")):
        for y in (470, 610, 750):
            draw.line((120, y, 960, y), fill=(70, 78, 90), width=8)
        draw.line((150, 1040, 930, 1040), fill=metal, width=12)
        for x in range(190, 900, 140):
            draw.line((x, 980, x + 70, 980), fill=warm, width=6)
    elif any(word in description for word in ("больниц", "hospital", "операцион", "ward")):
        draw.rectangle((170, 400, 910, 1080), fill=(205, 215, 216),
                       outline=(75, 88, 98), width=8)
        draw.rectangle((230, 490, 850, 570), fill=(35, 105, 105))
        draw.text((540, 530), "ПАЛАТА 13", font=get_font(38), fill="white", anchor="mm")
        draw.line((300, 850, 820, 850), fill=(90, 100, 105), width=8)
        draw.ellipse((460, 760, 620, 920), outline=accent, width=8)
    elif any(word in description for word in ("музей", "museum", "восков", "gallery")):
        for x in (210, 540, 870):
            draw.rectangle((x - 70, 500, x + 70, 1010), fill=(48, 42, 42),
                           outline=(180, 155, 110), width=6)
            draw.ellipse((x - 38, 560, x + 38, 710), fill=(105, 108, 112))
            draw.rectangle((x - 42, 710, x + 42, 940), fill=(70, 72, 78))
    elif any(word in description for word in ("подвод", "submarine", "корабл", "ship", "кают")):
        draw.ellipse((170, 430, 910, 1150), outline=metal, width=12)
        for x in (330, 540, 750):
            draw.ellipse((x - 48, 600, x + 48, 696), fill=(20, 35, 48),
                         outline=warm, width=6)
        draw.line((300, 900, 780, 900), fill=metal, width=10)
    else:
        offset = int(18 * math.sin(phase))
        draw.rectangle((90 + offset, 1030, 990 + offset, 1080),
                       fill=shadow, outline=accent, width=3)
        for x in range(150 + offset, 950 + offset, 160):
            draw.line((x, 1030, x + 30, 940), fill=(90, 98, 108), width=5)


def draw_scene(scene: dict[str, Any], story_meta: dict[str, Any], phase: float,

                             fill=(165, 205, 230), outline=(225, 240, 255))

            draw.polygon([(900, 1450), (820, 960), (730, 980), (790, 1450)],
                         fill=(56, 118, 182), outline=(20, 24, 30), width=5)

            draw.rounded_rectangle((290, 630, 790, 910), radius=32, fill=accent)
            draw.text((540, 770), str(scene.get("phone_header", "112 · SOS")).upper()[:20],

        elif template == "radio_kit":
            draw.rounded_rectangle((140, 430, 940, 1120), radius=34,
                                   fill=(38, 42, 48), outline=(190, 195, 205), width=10)
            draw.rectangle((220, 560, 860, 680), fill=(12, 16, 20),
                           outline=(225, 175, 65), width=5)
            draw.text((540, 620), str(scene.get("radio_frequency", "87.4 FM")),
                      font=get_font(58), fill=(245, 205, 95), anchor="mm")

            draw.ellipse((475, 1030, 605, 1160), fill=accent, outline="white", width=5)
        elif template in ("map_plan", "building_pipes"):
            draw.rectangle((130, 360, 950, 1180), fill=(20, 26, 36),
                           outline=(225, 230, 240), width=12)
            draw.line((470, 360, 470, 1180), fill=(225, 230, 240), width=10)
            draw.line((130, 800, 950, 800), fill=(225, 230, 240), width=10)
            draw.rectangle((150, 820, 440, 1150), fill=(24, 78, 52))
            draw.ellipse((650 - phase * 40, 700, 710 - phase * 40, 760),
                         fill=(55, 135, 225), outline="white", width=5)

                                   fill=(58, 62, 68), outline=(22, 26, 32), width=8)

            continue
        kind = command.get("op", "rect")
        dx = int(command.get("dx", 0)) * phase
        dy = int(command.get("dy", 0)) * phase
        fill = _rgb(command.get("fill")) if "fill" in command else None
        outline = _rgb(command.get("outline")) if "outline" in command else None
        stroke_width = int(command.get("w_stroke", 4))

        if kind == "rect":
            x1, y1 = int(command.get("x", 100)) + dx, int(command.get("y", 400)) + dy
            x2 = x1 + int(command.get("w", 200))
            y2 = y1 + int(command.get("h", 200))
            draw.rounded_rectangle((x1, y1, x2, y2), radius=int(command.get("radius", 12)),
                                   fill=fill, outline=outline,
                                   width=stroke_width if outline else 0)
        elif kind == "circle":
            cx, cy = int(command.get("x", 540)) + dx, int(command.get("y", 850)) + dy
            radius = int(command.get("r", 80))
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius),
                         fill=fill, outline=outline, width=stroke_width if outline else 0)
        elif kind == "line":
            draw.line((int(command.get("x1", 200)) + dx, int(command.get("y1", 600)) + dy,
                       int(command.get("x2", 800)) + dx, int(command.get("y2", 900)) + dy),
                      fill=fill or (235, 185, 65), width=int(command.get("w_stroke", 8)))
        elif kind == "actor":
            _draw_actor(draw, command, dx, dy)
        elif kind == "phone_ui":
            px, py = 270 + dx, 360 + dy
            draw.rounded_rectangle((px, py, px + 540, py + 940), radius=54,
                                   fill=(18, 22, 30), outline=(85, 92, 105), width=8)
            header = str(command.get("header", "112 · ЭКСТРЕННЫЙ ВЫЗОВ"))[:24]
            subtext = str(command.get("subtext", "Геопозиция отправлена"))[:32]
            badge_color = _rgb(command.get("fill"), (205, 45, 45))
            draw.rounded_rectangle((px + 45, py + 180, px + 495, py + 440),
                                   radius=28, fill=badge_color)
            draw.text((px + 270, py + 275), header.upper(), font=get_font(40),
                      fill="white", anchor="mm")
            draw.text((px + 270, py + 355), subtext, font=get_font(30),
                      fill=(235, 240, 245), anchor="mm")
        elif kind == "callout":
            cx, cy = int(command.get("x", 540)) + dx, int(command.get("y", 520)) + dy
            tx, ty = int(command.get("tx", cx)), int(command.get("ty", cy + 140))
            label = str(command.get("label", "ДЕТАЛЬ")).upper()[:26]
            tag_color = _rgb(command.get("fill"), (215, 52, 48))
            draw.line((cx, cy, tx, ty), fill=tag_color, width=5)
            draw.ellipse((tx - 9, ty - 9, tx + 9, ty + 9), fill=tag_color)
            text_width = int(draw.textlength(label, font=get_font(32))) + 40
            draw.rounded_rectangle((cx - text_width // 2, cy - 28, cx + text_width // 2, cy + 28),
                                   radius=12, fill=tag_color)
            draw.text((cx, cy), label, font=get_font(32), fill="white", anchor="mm")

    rec_title = str(story_meta.get("series_tag", "ЗАПИСЬ № 01")).upper()[:26]
    location = str(story_meta.get("location", "")).upper()[:34]
    clock = str(scene.get("clock", "02:17"))
    bpm = int(scene.get("bpm", 125)) + phase_step * 2
    timer_label = str(scene.get("timer_label", "")).strip()
    draw.ellipse((42, 52, 66, 76), fill=(230, 45, 45) if phase_step % 2 == 0 else (110, 25, 25))
    draw.text((82, 64), f"REC   {rec_title}", font=get_font(28),
              fill=(215, 220, 228), anchor="lm")
    if location:
        draw.text((540, 64), location, font=get_font(24),
                  fill=(150, 160, 172), anchor="mm")
    draw.text((WIDTH - 45, 64), clock, font=get_font(30), fill=(235, 115, 45), anchor="rm")
    draw.rounded_rectangle((WIDTH - 255, 105, WIDTH - 42, 168), radius=12,
                           fill=(14, 17, 22), outline=(210, 50, 50), width=3)
    draw.text((WIDTH - 230, 136), "♥", font=get_font(34), fill=(225, 50, 50), anchor="lm")
    draw.text((WIDTH - 60, 136), str(bpm), font=get_font(36), fill="white", anchor="rm")
    if timer_label:
        draw.rounded_rectangle((42, 105, 285, 168), radius=12,
                               fill=(14, 17, 22), outline=(235, 115, 45), width=3)
        draw.text((163, 136), timer_label.upper()[:18], font=get_font(30),
                  fill="white", anchor="mm")

    rule_badge = str(scene.get("rule_badge", "")).strip()
    visual_detail = str(scene.get("visual_detail", "")).strip()
    if rule_badge or visual_detail:
        header = f"{rule_badge} · {visual_detail}".strip(" ·").upper()[:36]
        box_width = max(int(draw.textlength(header, font=get_font(36))) + 64, 360)
        draw.rounded_rectangle((540 - box_width // 2, 215, 540 + box_width // 2, 289),
                               radius=14, fill=(18, 22, 28), outline=accent, width=4)
        draw.text((540, 252), header, font=get_font(36), fill="white", anchor="mm")
    environment_detail = str(scene.get("environment_detail", "")).strip()
    if environment_detail:
        draw.text((540, 315), environment_detail.upper()[:48],
                  font=get_font(24), fill=(190, 198, 208), anchor="mm")

    if show_choice is None:
        show_choice = bool(scene.get("is_ending") or scene.get("is_choice_scene"))
    choice_data = choice_source or scene
    if show_choice:
        choice_a = str(choice_data.get("choice_a") or "УБЕЖАТЬ").upper()[:16]
        choice_b = str(choice_data.get("choice_b") or "ОСТАТЬСЯ").upper()[:16]
        question = "ЧТО БЫ СДЕЛАЛ ТЫ? НАПИШИ В КОММЕНТАРИЯХ"
        question_width = min(
            WIDTH - 80,
            int(draw.textlength(question, font=get_font(30))) + 40,
        )
        draw.rounded_rectangle(
            (540 - question_width // 2, 970, 540 + question_width // 2, 1030),
            radius=10, fill=(14, 18, 24), outline=accent, width=3,
        )
        draw.text((540, 1000), question, font=get_font(30),
                  fill=(255, 205, 95), anchor="mm")
        for box, label, color in (
            ((100, 1060, 510, 1220), f"1. {choice_a}", (215, 48, 48)),
            ((570, 1060, 980, 1220), f"2. {choice_b}", (42, 145, 95)),
        ):
            draw.rounded_rectangle(box, radius=20, fill=color, outline="white", width=4)
            draw.text(((box[0] + box[2]) // 2, (box[1] + box[3]) // 2),
                      label, font=get_font(42), fill="white", anchor="mm")
        timer_value = max(1, 3 - (phase_step % 3))
        draw.rounded_rectangle((440, 1250, 640, 1310), radius=15, fill=accent)
        draw.text((540, 1280), f"ГОЛОСУЙ: {timer_value} СЕК",
                  font=get_font(32), fill="white", anchor="mm")
        clock_cx, clock_cy, clock_r = 735, 1280, 36
        draw.ellipse((clock_cx - clock_r, clock_cy - clock_r,
                      clock_cx + clock_r, clock_cy + clock_r),
                     fill=(18, 22, 28), outline=(255, 205, 95), width=4)
        angle = (phase * math.tau) - math.pi / 2
        hand_x = clock_cx + int(math.cos(angle) * 25)
        hand_y = clock_cy + int(math.sin(angle) * 25)
        draw.line((clock_cx, clock_cy, hand_x, hand_y), fill="white", width=5)
        draw.ellipse((clock_cx - 5, clock_cy - 5, clock_cx + 5, clock_cy + 5),
                     fill=(255, 205, 95))

    callout = str(scene.get("callout", "")).strip()
    if callout and not show_choice:
        label = callout.upper()[:34]
        draw.rounded_rectangle((70, 1080, 1010, 1150), radius=16,
                               fill=(18, 22, 28), outline=accent, width=3)
        draw.text((540, 1115), label, font=get_font(30), fill=(255, 220, 135),
                  anchor="mm")

    if words:
        lines: list[list[tuple[int, str]]] = [[]]
        for index, word in enumerate(words):
            if sum(len(item[1]) + 1 for item in lines[-1]) + len(word) > 20 and lines[-1]:
                lines.append([])
            lines[-1].append((index, word))
        font_sub = get_font(58)
        base_y = 1360
        for line in lines[-3:]:
            widths = [int(draw.textlength(word.upper(), font=font_sub)) for _, word in line]
            spacing = 18
            start_x = 540 - (sum(widths) + spacing * (len(line) - 1)) // 2
            for (word_index_value, word), width in zip(line, widths):
                color = (245, 130, 45) if word_index_value == word_index else "white"
                draw.text((start_x + width // 2, base_y), word.upper(), font=font_sub,
                          fill=color, anchor="mm", stroke_width=6, stroke_fill=(8, 10, 14))
                start_x += width + spacing
            base_y += 82
    return image


def generate_story(
    state_override: dict[str, Any] | None = None,
    final_for_video: bool = False,
) -> dict[str, Any]:
    state_file = BASE_DIR / "series_state.json"
    state: dict[str, Any] = dict(state_override or {})
    if state_override is None and state_file.exists():
        try:
            state = json.loads(state_file.read_text(encoding="utf-8"))
        except Exception:
            state = {}

    if "story_id" not in state:
        state["story_id"] = random.randint(10, 99)
    selected_choice = os.getenv("VIEWER_CHOICE", "").strip().lower()
    if selected_choice in {"a", "b", "1", "2"}:
        selected_choice = "a" if selected_choice in {"a", "1"} else "b"
        state["selected_choice"] = selected_choice
    ep = int(state.get("episode", 0)) + 1
    tot = int(state.get("total_episodes", 0))
    if ep > tot or tot < 3:
        tot = random.randint(3, 5)
        ep = 1
        state = {
            "story_id": random.randint(10, 99),
            "title": "",
            "episode": 1,
            "total_episodes": tot,
            "prev_cliffhanger": ""
        }

    print(f"🤖 [1/5] ИИ генерирует уникальный сериал (Эпизод {ep} из {tot})...")
    recent_locations = state.get("recent_locations", [])
    recent_titles = state.get("recent_titles", [])
    continuity = (
        f"Это СЕРИЯ {ep} из {tot}. Первая сцена ОБЯЗАНА прямо продолжать финал прошлой серии: "
        f"«{state.get('prev_cliffhanger', '')}». Возможные направления начала: "
        f"{state.get('possible_openings', [])}. Предыдущие ветки выбора: "
        f"{state.get('choice_branches', {})}. Выбор зрителей: "
        f"{state.get('selected_choice', 'ещё не выбран')}."
        if ep > 1 else
        f"Это СЕРИЯ 1 из {tot}. Придумай СОВЕРШЕННО СЕКРЕТНУЮ, жуткую и оригинальную тему выживания с нуля. "
        "Никаких повторов — выбери уникальную локацию."
    )
    exclusion = (
        f"Не используй повторно эти локации: {', '.join(map(str, recent_locations[-8:]))}. "
        f"Не повторяй эти названия: {', '.join(map(str, recent_titles[-8:]))}."
        if recent_locations or recent_titles else
        "Выбери локацию и название, которых раньше не было."
    )
    ending_instruction = (
        "Это финал всего выпуска. Только в 9-й сцене задай вопрос «А что бы сделал ты?» "
        "и заполни choice_a и choice_b."
        if final_for_video else
        "Это промежуточный акт выпуска. Не добавляй голосование и вопрос зрителю; "
        "закончи сильным клиффхэнгером для следующего акта."
    )
    prompt = f"""
Ты — гениальный хоррор-шоураннер, который создает вирусные многосерийные истории для TikTok и Shorts.
{continuity}
{exclusion}
{ending_instruction}
Уникальный seed выпуска: {random.randint(100000, 999999)}.
Каждая сцена должна иметь визуально конкретный фон, соответствующий действию: укажи
environment_detail (3-8 слов), background_scene (5-12 слов) и

def make_scene_clip(scene: dict[str, Any], story_meta: dict[str, Any], audio_path: Path | None,
                    timestamp: str, scene_number: int,
                    clips: list[Any], audio_clips: list[Any],
                    duration_scale: float = 1.0,
                    absolute_start: float = 0.0,
                    choice_overlay_start: list[float] | None = None,
                    choice_source: dict[str, Any] | None = None) -> None:
    audio = None
    if audio_path is not None:
        source_audio = AudioFileClip(str(audio_path))
        audio_clips.append(source_audio)
        target_duration = get_scene_duration(scene) * max(duration_scale, 0.1)
        if abs(source_audio.duration - target_duration) > 0.05:
            audio = source_audio.with_speed_scaled(
                factor=max(source_audio.duration / target_duration, 0.1)
            )
            audio_clips.append(audio)
        else:
            audio = source_audio
    else:
        scene_duration_for_clip = get_scene_duration(scene) * max(duration_scale, 0.1)

    words = str(scene.get("text", "")).split()
    duration = max(float(audio.duration), 0.1) if audio is not None else scene_duration_for_clip
    if not words:
        words = [""]
    word_duration = duration / max(len(words), 1)
    word_clips = []
    for index, _ in enumerate(words):
        frame = np.asarray(draw_scene(scene, story_meta, 0.0, index, words,
                                      show_choice=False), dtype=np.uint8)
        if frame.shape != (HEIGHT, WIDTH, 3) or not frame.any():
            raise RuntimeError(f"Пустой кадр сцены {scene_number}, слово {index}")
        segment_duration = (
            word_duration
            if index < len(words) - 1
            else duration - word_duration * (len(words) - 1)
        )
        segment_start = index * word_duration

        def make_frame(
            t: float,
            word_index: int = index,
            clip_duration: float = segment_duration,
            start_time: float = segment_start,
        ) -> np.ndarray:
            progress = min(max(float(t) / max(clip_duration, 0.001), 0.0), 1.0)
            motion_phase = ((start_time + float(t)) / duration) * 4.0
            absolute_time = absolute_start + start_time + float(t)
            show_choice = (
                choice_overlay_start is not None
                and absolute_time >= choice_overlay_start[0]
                and choice_source is not None
            )
            return np.asarray(
                draw_scene(scene, story_meta, motion_phase, word_index, words,
                           show_choice=show_choice, choice_source=choice_source),
                dtype=np.uint8,
            )

        word_clips.append(VideoClip(
            frame_function=make_frame,
            duration=max(segment_duration, 0.001),
        ))
    base_clip = concatenate_videoclips(word_clips, method="compose")
    # Keep video strictly inside the audio interval; never extend either side.
    scene_duration = min(base_clip.duration, duration)
    if base_clip.duration > scene_duration:
        base_clip = base_clip.subclipped(0, scene_duration)
    scene_clip = base_clip.with_audio(audio) if audio is not None else base_clip
    clips.append(scene_clip)

import os
import pickle
from pathlib import Path
from google_auth_oauthlib.flow import InstalledAppFlow
from googleapiclient.discovery import build
from googleapiclient.http import MediaFileUpload
from google.auth.transport.requests import Request

SCOPES = ["https://www.googleapis.com/auth/youtube.upload"]
BASE_DIR = Path(__file__).resolve().parent.parent
SCRIPT_DIR = Path(__file__).resolve().parent


def get_authenticated_service():
    """Авторизуется в YouTube API (при первом запуске откроет браузер для входа)"""
    creds = None
    token_path = BASE_DIR / "token.pickle"
    client_secret_path = BASE_DIR / "client_secret.json"

    if token_path.exists():
        with open(token_path, "rb") as token:
            creds = pickle.load(token)

    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            if not client_secret_path.exists():
                client_secret_path = SCRIPT_DIR / "client_secret.json"
            if not client_secret_path.exists():
                raise FileNotFoundError(
                    f"❌ Не найден client_secret.json в {BASE_DIR} или {SCRIPT_DIR}."
                )
            flow = InstalledAppFlow.from_client_secrets_file(str(client_secret_path), SCOPES)
            creds = flow.run_local_server(port=0)

        with open(token_path, "wb") as token:
            pickle.dump(creds, token)

    return build("youtube", "v3", credentials=creds)


def upload_short(video_path: str, title: str, description: str = "#shorts #ai #quiz #makeiteasyai"):
    """Загружает готовый файл на YouTube"""
    youtube = get_authenticated_service()

    body = {
        "snippet": {
            "title": title,
            "description": description,
            "tags": ["shorts", "ai", "quiz", "makeiteasyai"],
            "categoryId": "28"  # Наука и технологии
        },
        "status": {
            "privacyStatus": os.getenv("YOUTUBE_PRIVACY_STATUS", "private"),
            "selfDeclaredMadeForKids": False
        }
    }

    media = MediaFileUpload(video_path, chunksize=-1, resumable=True)

    print(f"📤 Заливаем на YouTube: {title}...")
    request = youtube.videos().insert(
        part="snippet,status",
        body=body,
        media_body=media
    )

    response = None
    while response is None:
        status, response = request.next_chunk()
        if status:
            print(f"⏳ Загрузка: {int(status.progress() * 100)}%")

    print(f"🎉 Успешно залито! Ссылка: https://youtu.be/{response['id']}")
    return response['id']
